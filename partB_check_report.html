Reproduction

library(tidyverse)
library(readxl)

data_dir <- "partB_panrivers/data"

# Experiment 3 only (sheet names per the archive details file)
preds_raw <- read_excel(
  file.path(data_dir, "Pan & Rivers (2022) - KUPE Globally Differentiated Predictions Data.xlsx"),
  sheet = "kupe3gpredsrfile"
)
crit_raw <- read_excel(
  file.path(data_dir, "Pan & Rivers (2022) - KUPE Criterial Test Data.xlsx"),
  sheet = "kupe3finaltest"
)
# Subject-level data: one row per subject x practice condition x cycle.
# Predictions code the condition as "Pretest"/"Read"; the criterial test as "Pretested"/"Read".
preds <- preds_raw |>
  mutate(COND = recode(COND, Pretest = "Pretested"))

# Sanity checks: N = 52, balanced 2 x 2 within-subject design, no missing values
stopifnot(
  n_distinct(preds$SUBJ) == 52,
  n_distinct(crit_raw$SUBJ) == 52,
  nrow(preds) == 52 * 4,
  nrow(crit_raw) == 52 * 4,
  !anyNA(preds), !anyNA(crit_raw)
)

# Wide format: one column per condition x cycle cell
preds_wide <- preds |>
  pivot_wider(id_cols = SUBJ, names_from = c(COND, CYCLE), values_from = PREDICTION)
crit_wide <- crit_raw |>
  pivot_wider(id_cols = SUBJ, names_from = c(COND, CYCLE), values_from = SCORE)
# In a 2 x 2 fully within-subject design every effect has 1 numerator df, so the
# repeated-measures ANOVA F for an effect equals the squared one-sample t on the
# per-subject contrast for that effect (F = t^2, same p, same df).
# Partial eta squared = F * df1 / (F * df1 + df2) = t^2 / (t^2 + df2).
contrast_F <- function(x) {
  tt <- t.test(x)
  df2 <- unname(tt$parameter)
  F <- unname(tt$statistic)^2
  tibble(F = F, df1 = 1, df2 = df2, p = tt$p.value, np2 = F / (F + df2))
}

# Outcome 1: cycle x practice condition interaction on predictions
# (Pretested - Read) in cycle TWO minus the same difference in cycle ONE
out1 <- contrast_F(with(preds_wide,
  (Pretested_TWO - Read_TWO) - (Pretested_ONE - Read_ONE)))

# Outcome 2: main effect of cycle on predictions (TWO - ONE, averaged over conditions)
out2 <- contrast_F(with(preds_wide,
  (Pretested_TWO + Read_TWO) / 2 - (Pretested_ONE + Read_ONE) / 2))

# Outcomes 3 and 4: criterial test, pretested vs read, paired t-test within each cycle
paired_t <- function(pretested, read) {
  diffs <- pretested - read
  tt <- t.test(pretested, read, paired = TRUE)
  tibble(t = unname(tt$statistic), df = unname(tt$parameter),
         p = tt$p.value, d = mean(diffs) / sd(diffs))   # d = mean diff / SD of diffs
}
out3 <- paired_t(crit_wide$Pretested_ONE, crit_wide$Read_ONE)
out4 <- paired_t(crit_wide$Pretested_TWO, crit_wide$Read_TWO)
# Reproduced values next to the paper values
results <- tribble(
  ~outcome, ~statistic, ~paper, ~reproduced,
  "1. Cycle x condition interaction (predictions)", "F(1, 51)", 10.92, out1$F,
  "1. Cycle x condition interaction (predictions)", "p",         0.0017, out1$p,
  "1. Cycle x condition interaction (predictions)", "partial eta^2", 0.18, out1$np2,
  "2. Main effect of cycle (predictions)",          "F(1, 51)", 8.83, out2$F,
  "2. Main effect of cycle (predictions)",          "p",         0.0045, out2$p,
  "2. Main effect of cycle (predictions)",          "partial eta^2", 0.15, out2$np2,
  "3. Criterial test, cycle 1",                     "t(51)",     6.67, out3$t,
  "3. Criterial test, cycle 1",                     "p",         NA, out3$p,   # paper: p < .00001
  "3. Criterial test, cycle 1",                     "d",         0.93, out3$d,
  "4. Criterial test, cycle 2",                     "t(51)",     5.99, out4$t,
  "4. Criterial test, cycle 2",                     "p",         NA, out4$p,   # paper: p < .0001

)
